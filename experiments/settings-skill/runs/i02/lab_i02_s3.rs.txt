//! Gallery scenario I02/S3: localized General settings with long Japanese copy.
use gpui::{prelude::*, AnyElement, App, Window};
use gpui_component::IconName;
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::i18n::Msg;
use kagi::ui::settings_kit::{
    detail_page, group, icon_tile, popup_value, section, settings_window, sidebar, sidebar_gap,
    sidebar_item, stepper, switch, Forced, Row, Segmented, StepperPart, Tint,
};

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i02-s3",
    states: &["default", "focus-ui-zoom"],
    size: (900., 692.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn pane_item(id: &'static str, icon: IconName, label: Msg, selected: bool) -> AnyElement {
    sidebar_item(
        id,
        icon_tile(icon, Tint::Accent, 20.),
        label.t(),
        None,
        selected,
        Forced::Rest,
    )
    .into_any_element()
}

fn general_sidebar() -> AnyElement {
    sidebar([
        pane_item("general", IconName::Settings2, Msg::SettingsGeneral, true),
        pane_item(
            "language",
            IconName::ALargeSmall,
            Msg::SettingsLanguage,
            false,
        ),
        pane_item(
            "smart-commit",
            IconName::Bot,
            Msg::SettingsSmartCommit,
            false,
        ),
        pane_item("analyze", IconName::ChartPie, Msg::SettingsAnalyze, false),
        sidebar_gap(),
    ])
    .into_any_element()
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let zoom_focus =
        (cx.state == "focus-ui-zoom").then_some((StepperPart::Increase, Forced::Focus));
    settings_window(
        general_sidebar(),
        detail_page(
            "general-scroll",
            Msg::SettingsGeneral.t(),
            [
                section(
                    Msg::SettingsAppearance.t(),
                    None,
                    None,
                    group([
                        Row::new(Msg::SettingsTheme.t())
                            .subtitle(Msg::SettingsThemeDesc.t())
                            .trailing(popup_value("theme", "Apple Dark", Forced::Rest))
                            .into_any_element(),
                        Row::new(Msg::SettingsAccent.t())
                            .trailing(
                                Segmented::new(
                                    "accent",
                                    [Msg::SettingsAccentSystem.t(), Msg::SettingsAccentTheme.t()],
                                    1,
                                )
                                .forced(1, Forced::Rest),
                            )
                            .into_any_element(),
                        Row::new(Msg::SettingsToolbarDisplay.t())
                            .subtitle(Msg::SettingsToolbarDisplayDesc.t())
                            .trailing(
                                Segmented::new(
                                    "toolbar",
                                    [Msg::ToolbarIconAndText.t(), Msg::ToolbarIconOnly.t()],
                                    1,
                                )
                                .forced(1, Forced::Rest),
                            )
                            .into_any_element(),
                        Row::new(Msg::SettingsZoom.t())
                            .subtitle(Msg::SettingsZoomDesc.t())
                            .trailing(stepper("ui-zoom", "100%", true, true, zoom_focus))
                            .into_any_element(),
                        Row::new(Msg::SettingsCompactThreshold.t())
                            .subtitle(Msg::SettingsCompactThresholdDesc.t())
                            .trailing(stepper("compact-threshold", "70rem", true, true, None))
                            .into_any_element(),
                    ]),
                ),
                section(
                    Msg::SettingsGraph.t(),
                    None,
                    None,
                    group([
                        Row::new(Msg::SettingsCompact.t())
                            .subtitle(Msg::SettingsCompactDesc.t())
                            .trailing(switch(
                                "compact-graph",
                                Msg::SettingsCompact.t(),
                                false,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new(Msg::SettingsLaneCompact.t())
                            .subtitle(Msg::SettingsLaneCompactDesc.t())
                            .trailing(switch(
                                "avatar-commit-nodes",
                                Msg::SettingsLaneCompact.t(),
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                    ]),
                ),
                section(
                    Msg::SettingsNetwork.t(),
                    None,
                    None,
                    group([Row::new(Msg::SettingsAutoFetch.t())
                        .subtitle(Msg::SettingsAutoFetchDesc.t())
                        .trailing(switch(
                            "auto-fetch",
                            Msg::SettingsAutoFetch.t(),
                            true,
                            Forced::Rest,
                            |_, _, _| {},
                        ))
                        .into_any_element()]),
                ),
            ],
        ),
    )
    .into_any_element()
}
