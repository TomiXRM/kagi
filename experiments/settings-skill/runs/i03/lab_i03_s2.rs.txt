//! Gallery reconstruction of Rex's Interface settings pane (I03/S2).

use gpui::{div, prelude::*, rgb, AnyElement, App, FontWeight, Role, Window};
use gpui_component::IconName;
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::settings_kit::{
    caption, detail_page, group, icon_tile, ink, popup_value, pt, ring, search_field, section,
    settings_window, sidebar, sidebar_gap, sidebar_item, switch, vertical_divider, Forced, Row,
    Segmented, Tint,
};

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i03-s2",
    states: &["default", "focus-slider", "pressed-choice"],
    size: (900., 692.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn rex_sidebar() -> AnyElement {
    let item = |id: &'static str, icon, label: &'static str, selected: bool| {
        sidebar_item(
            id,
            icon_tile(
                icon,
                if selected {
                    Tint::Accent
                } else {
                    Tint::Neutral
                },
                20.,
            ),
            label,
            None,
            selected,
            Forced::Rest,
        )
        .into_any_element()
    };

    sidebar([
        search_field("search", "Search Settings").into_any_element(),
        sidebar_item(
            "server",
            icon_tile(IconName::Bot, Tint::Success, 20.),
            "Server",
            Some("Running".into()),
            false,
            Forced::Rest,
        )
        .into_any_element(),
        sidebar_gap(),
        item("interface", IconName::LayoutDashboard, "Interface", true),
        item("themes", IconName::Palette, "Themes", false),
        sidebar_gap(),
        item("terminal", IconName::SquareTerminal, "Terminal", false),
        item("shortcuts", IconName::Settings2, "Shortcuts", false),
        item("integrations", IconName::Github, "Integrations", false),
        item("notifications", IconName::Bell, "Notifications", false),
        item("remote-hosts", IconName::Globe, "Remote Hosts", false),
        sidebar_gap(),
        item("developer", IconName::Settings, "Developer", false),
    ])
    .into_any_element()
}

/// Page-local kit gap: a visual-choice tile. The kit has text segmented
/// controls, but these choices need a visual preview and a label beneath it.
fn choice_tile(
    id: &'static str,
    preview: AnyElement,
    label: &'static str,
    selected: bool,
    forced: Forced,
) -> AnyElement {
    let fill = match forced {
        Forced::Hover => ink::control_hover(),
        Forced::Pressed => ink::control_pressed(),
        _ => ink::control(),
    };
    let preview = div()
        .w(pt(58.))
        .h(pt(54.))
        .flex()
        .items_center()
        .justify_center()
        .rounded(pt(8.))
        .bg(rgb(fill))
        .child(preview);
    let tile = div()
        .id(id)
        .w(pt(62.))
        .flex()
        .flex_col()
        .items_center()
        .gap(pt(6.))
        .role(Role::RadioButton)
        .aria_label(label)
        .aria_selected(selected)
        .child(preview)
        .child(
            div()
                .text_color(if selected {
                    ink::text()
                } else {
                    ink::secondary()
                })
                .when(selected, |el| el.font_weight(FontWeight::SEMIBOLD))
                .child(label),
        )
        .when(forced == Forced::Disabled, |el| {
            el.opacity(ink::disabled_opacity())
        });
    ring(tile, selected || forced == Forced::Focus, 10.).into_any_element()
}

fn window_preview(light: bool) -> AnyElement {
    let page = if light { ink::card() } else { ink::window() };
    div()
        .w(pt(30.))
        .h(pt(34.))
        .rounded(pt(6.))
        .bg(rgb(page))
        .border_1()
        .border_color(rgb(ink::hairline()))
        .child(div().h(pt(7.)).rounded_t(pt(6.)).bg(rgb(ink::control())))
        .child(
            div()
                .p(pt(4.))
                .flex()
                .flex_col()
                .gap(pt(3.))
                .child(div().h(pt(8.)).rounded(pt(2.)).bg(rgb(ink::control())))
                .child(
                    div()
                        .flex()
                        .gap(pt(3.))
                        .child(
                            div()
                                .flex_1()
                                .h(pt(7.))
                                .rounded(pt(2.))
                                .bg(rgb(ink::control())),
                        )
                        .child(
                            div()
                                .flex_1()
                                .h(pt(7.))
                                .rounded(pt(2.))
                                .bg(rgb(ink::control())),
                        ),
                ),
        )
        .into_any_element()
}

fn style_preview(glass: bool) -> AnyElement {
    div()
        .size(pt(34.))
        .rounded_full()
        .bg(rgb(if glass {
            ink::control_hover()
        } else {
            ink::window()
        }))
        .border_1()
        .border_color(rgb(ink::hairline()))
        .into_any_element()
}

fn appearance_choices(choice_forced: Forced) -> AnyElement {
    let scheme = div()
        .flex_1()
        .min_w_0()
        .flex()
        .flex_col()
        .gap(pt(8.))
        .child(caption("Color scheme"))
        .child(
            div()
                .flex()
                .justify_between()
                .child(choice_tile(
                    "scheme-auto",
                    window_preview(false),
                    "Auto",
                    true,
                    choice_forced,
                ))
                .child(choice_tile(
                    "scheme-light",
                    window_preview(true),
                    "Light",
                    false,
                    Forced::Rest,
                ))
                .child(choice_tile(
                    "scheme-dark",
                    window_preview(false),
                    "Dark",
                    false,
                    Forced::Rest,
                )),
        );
    let style = div()
        .flex_1()
        .min_w_0()
        .flex()
        .flex_col()
        .gap(pt(8.))
        .child(caption("Interface style"))
        .child(
            div()
                .flex()
                .justify_between()
                .child(choice_tile(
                    "style-glass",
                    style_preview(true),
                    "Glass",
                    false,
                    Forced::Rest,
                ))
                .child(choice_tile(
                    "style-classic",
                    style_preview(false),
                    "Classic",
                    true,
                    Forced::Rest,
                )),
        );
    div()
        .min_h(pt(112.))
        .py(pt(10.))
        .flex()
        .items_stretch()
        .gap(pt(12.))
        .child(scheme)
        .child(vertical_divider())
        .child(style)
        .into_any_element()
}

/// Page-local kit gap: a compact, five-position slider with a trailing value.
fn ticked_slider(
    id: &'static str,
    label: &'static str,
    value: usize,
    tick_count: usize,
    forced: Forced,
) -> AnyElement {
    let track_width = 132.;
    let fraction =
        (value.saturating_sub(1) as f32 / (tick_count.saturating_sub(1)) as f32).clamp(0., 1.);
    let filled = track_width * fraction;
    let track = div()
        .id(id)
        .w(pt(track_width))
        .h(pt(24.))
        .relative()
        .flex()
        .items_center()
        .role(Role::Slider)
        .aria_label(label)
        .aria_numeric_value(value as f64)
        .aria_min_numeric_value(1.)
        .aria_max_numeric_value(tick_count as f64)
        .child(
            div().absolute().inset_0().flex().items_center().child(
                div()
                    .w_full()
                    .h(pt(4.))
                    .rounded_full()
                    .bg(rgb(ink::control())),
            ),
        )
        .child(
            div().absolute().inset_0().flex().items_center().child(
                div()
                    .w(pt(filled))
                    .h(pt(4.))
                    .rounded_full()
                    .bg(ink::accent()),
            ),
        )
        .child(
            div()
                .absolute()
                .inset_0()
                .flex()
                .items_center()
                .justify_between()
                .children((0..tick_count).map(|_| {
                    div()
                        .size(pt(3.))
                        .rounded_full()
                        .bg(rgb(ink::card()))
                        .into_any_element()
                })),
        )
        .child(
            div()
                .w(pt(filled))
                .h_full()
                .flex()
                .items_center()
                .justify_end()
                .child(
                    div()
                        .size(pt(12.))
                        .rounded_full()
                        .bg(ink::text())
                        .border_1()
                        .border_color(rgb(ink::hairline())),
                ),
        );
    div()
        .flex()
        .items_center()
        .gap(pt(10.))
        .child(ring(track, forced == Forced::Focus, 8.))
        .child(
            div()
                .min_w(pt(8.))
                .text_color(ink::secondary())
                .child(value.to_string()),
        )
        .into_any_element()
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let choice_forced = if cx.state == "pressed-choice" {
        Forced::Pressed
    } else {
        Forced::Rest
    };
    let slider_forced = if cx.state == "focus-slider" {
        Forced::Focus
    } else {
        Forced::Rest
    };
    settings_window(
        rex_sidebar(),
        detail_page(
            "interface-scroll",
            "Interface",
            [
                section(
                    "Appearance",
                    None,
                    None,
                    group([
                        appearance_choices(choice_forced),
                        Row::new("Tint with theme")
                            .trailing(switch(
                                "tint-with-theme",
                                "Tint with theme",
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                    ]),
                ),
                section(
                    "Tabs",
                    None,
                    None,
                    group([
                        Row::new("Tab style")
                            .trailing(
                                Segmented::new(
                                    "tab-style",
                                    ["Above", "Left Side", "Right Side"],
                                    0,
                                )
                                .forced(0, Forced::Rest),
                            )
                            .into_any_element(),
                        Row::new("Fill available space")
                            .trailing(switch(
                                "fill-space",
                                "Fill available space",
                                false,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new("Show tab icons")
                            .trailing(ticked_slider(
                                "show-tab-icons",
                                "Show tab icons",
                                3,
                                5,
                                slider_forced,
                            ))
                            .into_any_element(),
                        Row::new("Use color for tab icons")
                            .trailing(popup_value("tab-icon-color", "Always", Forced::Rest))
                            .into_any_element(),
                        Row::new("Shell icon theme")
                            .trailing(popup_value("shell-icon-theme", "Default", Forced::Rest))
                            .into_any_element(),
                    ]),
                ),
                section(
                    "Panes",
                    None,
                    None,
                    group([
                        Row::new("Pane style")
                            .trailing(
                                Segmented::new("pane-style", ["Comfortable", "Compact"], 0)
                                    .forced(0, Forced::Rest),
                            )
                            .into_any_element(),
                        Row::new("Show pane headers")
                            .trailing(switch(
                                "pane-headers",
                                "Show pane headers",
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                    ]),
                ),
            ],
        ),
    )
    .into_any_element()
}
