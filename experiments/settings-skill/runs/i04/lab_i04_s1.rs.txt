//! Gallery reconstruction of Rex's Notifications settings pane (I04/S1).
//!
//! The only page-local control is `position_picker`: settings_kit has no
//! two-dimensional notification-position picker yet.
use gpui::{div, prelude::*, rgb, AnyElement, App, Role, Window};
use gpui_component::{Icon, IconName, Sizable as _, Size};
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::settings_kit::{
    group, icon_tile, ink, pt, ring, search_field, section, settings_window, sidebar, sidebar_gap,
    sidebar_item, switch, Forced, Row, Tint,
};

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i04-s1",
    states: &[
        "default",
        "rest",
        "selected",
        "focus",
        "focus-picker",
        "hover-switch",
        "focus-nav",
    ],
    size: (900., 692.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn rex_sidebar() -> AnyElement {
    let item = |id: &'static str, icon, tint, label: &'static str, selected: bool| {
        sidebar_item(
            id,
            // The kit requires every selected sidebar tile to use Accent.
            icon_tile(icon, if selected { Tint::Accent } else { tint }, 20.),
            label,
            None,
            selected,
            Forced::Rest,
        )
        .into_any_element()
    };

    sidebar([
        search_field("search", "Search Settings").into_any_element(),
        sidebar_item(
            "server",
            icon_tile(IconName::Bot, Tint::Purple, 32.),
            "Server",
            Some("Running".into()),
            false,
            Forced::Rest,
        )
        .into_any_element(),
        sidebar_gap(),
        item(
            "interface",
            IconName::LayoutDashboard,
            Tint::Neutral,
            "Interface",
            false,
        ),
        item("themes", IconName::Palette, Tint::Accent, "Themes", false),
        sidebar_gap(),
        item(
            "terminal",
            IconName::SquareTerminal,
            Tint::Success,
            "Terminal",
            false,
        ),
        item(
            "shortcuts",
            IconName::Settings2,
            Tint::Neutral,
            "Shortcuts",
            false,
        ),
        item(
            "integrations",
            IconName::Github,
            Tint::Tag,
            "Integrations",
            false,
        ),
        item(
            "notifications",
            IconName::Bell,
            Tint::Accent,
            "Notifications",
            true,
        ),
        item(
            "remote-hosts",
            IconName::Globe,
            Tint::Remote,
            "Remote Hosts",
            false,
        ),
        sidebar_gap(),
        item(
            "developer",
            IconName::Settings2,
            Tint::Neutral,
            "Developer",
            false,
        ),
    ])
    .into_any_element()
}

/// Page-local kit gap: a reusable 3 by 2 notification placement selector.
/// `selected` is (column, row), with (2, 1) representing bottom right.
fn position_picker(selected: (usize, usize), forced: Forced) -> impl IntoElement {
    let names = [
        ["Top left", "Top center", "Top right"],
        ["Bottom left", "Bottom center", "Bottom right"],
    ];
    let mut rows = div().flex().flex_col().gap(pt(4.));
    for (row, labels) in names.into_iter().enumerate() {
        let mut cells = div().flex().gap(pt(4.));
        for (column, label) in labels.into_iter().enumerate() {
            let on = selected == (column, row);
            let cell = div()
                .id(format!("notification-position-{row}-{column}"))
                .size(pt(30.))
                .flex()
                .items_center()
                .justify_center()
                .rounded(pt(7.))
                .role(Role::RadioButton)
                .aria_label(label)
                .aria_selected(on)
                .when(on, |el| {
                    el.bg(ink::selected())
                        .child(Icon::new(IconName::Check).with_size(Size::Size(pt(13.))))
                })
                .when(!on, |el| el.bg(rgb(ink::control())));
            cells = cells.child(cell);
        }
        rows = rows.child(cells);
    }
    ring(
        div()
            .id("notification-position")
            .p(pt(5.))
            .rounded(pt(10.))
            .bg(rgb(ink::control()))
            .role(Role::RadioGroup)
            .aria_label("Notification position")
            .child(rows),
        forced == Forced::Focus,
        10.,
    )
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let selected_position = if cx.state == "selected" {
        (0, 1)
    } else {
        (2, 1)
    };
    let position_subtitle = if selected_position == (0, 1) {
        "Displayed bottom left"
    } else {
        "Displayed bottom right"
    };
    let picker_state = if matches!(cx.state, "focus" | "focus-picker") {
        Forced::Focus
    } else {
        Forced::Rest
    };
    let first_switch_state = if cx.state == "hover-switch" {
        Forced::Hover
    } else {
        Forced::Rest
    };
    let navigation_state = if cx.state == "focus-nav" {
        Forced::Focus
    } else {
        Forced::Rest
    };

    settings_window(
        rex_sidebar(),
        kagi::ui::settings_kit::detail_page(
            "notifications-scroll",
            "Notifications",
            [
                section(
                    "In-App Notifications",
                    None,
                    None,
                    group([Row::new("Notification position")
                        .subtitle(position_subtitle)
                        .trailing(position_picker(selected_position, picker_state))
                        .into_any_element()]),
                ),
                section(
                    "Program Activity",
                    Some("How status from processes and agents is handled".into()),
                    None,
                    group([
                        Row::new("Show activity in tabs")
                            .trailing(switch(
                                "show-activity",
                                "Show activity in tabs",
                                true,
                                first_switch_state,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new("Show system notifications")
                            .trailing(switch(
                                "show-system-notifications",
                                "Show system notifications",
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new("Badge Dock icon")
                            .trailing(switch(
                                "badge-dock-icon",
                                "Badge Dock icon",
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new("Play sounds")
                            .trailing(switch(
                                "play-sounds",
                                "Play sounds",
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                    ]),
                ),
                group([
                    Row::new("Working")
                        .subtitle("Tabs")
                        .leading(icon_tile(IconName::Ellipsis, Tint::Accent, 26.))
                        .navigation("working-notification", navigation_state)
                        .into_any_element(),
                    Row::new("Blocked")
                        .subtitle("Tabs, Notifications, Dock Badge, Sound")
                        .leading(icon_tile(IconName::CircleX, Tint::Warning, 26.))
                        .navigation("blocked-notification", Forced::Rest)
                        .into_any_element(),
                    Row::new("Error")
                        .subtitle("Tabs, Notifications, Dock Badge, Sound")
                        .leading(icon_tile(IconName::TriangleAlert, Tint::Danger, 26.))
                        .navigation("error-notification", Forced::Rest)
                        .into_any_element(),
                    Row::new("Done")
                        .subtitle("Tabs, Notifications, Dock Badge, Sound")
                        .leading(icon_tile(IconName::CircleCheck, Tint::Success, 26.))
                        .navigation("done-notification", Forced::Rest)
                        .into_any_element(),
                ])
                .into_any_element(),
                group([
                    Row::new("Include idle processes in activity")
                        .trailing(switch(
                            "include-idle-processes",
                            "Include idle processes in activity",
                            true,
                            Forced::Rest,
                            |_, _, _| {},
                        ))
                        .into_any_element(),
                    Row::new("Show each process in tabs")
                        .trailing(switch(
                            "show-each-process",
                            "Show each process in tabs",
                            false,
                            Forced::Rest,
                            |_, _, _| {},
                        ))
                        .into_any_element(),
                ])
                .into_any_element(),
            ],
        ),
    )
    .into_any_element()
}
