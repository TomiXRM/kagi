//! Gallery reconstruction for S3: the localized General settings pane.
use gpui::{prelude::*, AnyElement, App, Window};
use gpui_component::IconName;
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::settings_kit::{
    detail_page, group, icon_tile, popup_value, section, settings_window, sidebar, sidebar_gap,
    sidebar_item, stepper, switch, Forced, Row, Segmented, StepperPart, Tint,
};
use kagi_ui_core::i18n::Msg;

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i04-s3",
    states: &["default", "focus-auto-fetch"],
    size: (900., 692.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn settings_sidebar(focus: Forced) -> AnyElement {
    let item = |id, icon, label, selected| {
        sidebar_item(
            id,
            icon_tile(
                icon,
                if selected {
                    Tint::Accent
                } else {
                    Tint::Neutral
                },
                20.,
            ),
            label,
            None,
            selected,
            if id == "general" { Forced::Rest } else { focus },
        )
        .into_any_element()
    };

    sidebar([
        item(
            "general",
            IconName::Settings2,
            Msg::SettingsGeneral.t(),
            true,
        ),
        sidebar_gap(),
        item(
            "language",
            IconName::ALargeSmall,
            Msg::SettingsLanguage.t(),
            false,
        ),
        item(
            "smart-commit",
            IconName::Bot,
            Msg::SettingsSmartCommit.t(),
            false,
        ),
        item(
            "analyze",
            IconName::ChartPie,
            Msg::SettingsAnalyze.t(),
            false,
        ),
    ])
    .into_any_element()
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let auto_fetch_forced = if cx.state == "focus-auto-fetch" {
        Forced::Focus
    } else {
        Forced::Rest
    };

    settings_window(
        settings_sidebar(Forced::Rest),
        detail_page(
            "general-scroll",
            Msg::SettingsGeneral.t(),
            [
                section(
                    Msg::SettingsAppearance.t(),
                    None,
                    None,
                    group([
                        Row::new(Msg::SettingsTheme.t())
                            .subtitle(Msg::SettingsThemeDesc.t())
                            .trailing(popup_value("theme", "Apple Dark", Forced::Rest))
                            .into_any_element(),
                        Row::new(Msg::SettingsAccent.t())
                            .trailing(
                                Segmented::new(
                                    "accent",
                                    [Msg::SettingsAccentSystem.t(), Msg::SettingsAccentTheme.t()],
                                    1,
                                )
                                .forced(1, Forced::Rest),
                            )
                            .into_any_element(),
                        Row::new(Msg::SettingsToolbarDisplay.t())
                            .subtitle(Msg::SettingsToolbarDisplayDesc.t())
                            .trailing(
                                Segmented::new(
                                    "toolbar",
                                    [Msg::ToolbarIconAndText.t(), Msg::ToolbarIconOnly.t()],
                                    1,
                                )
                                .forced(1, Forced::Rest),
                            )
                            .into_any_element(),
                        Row::new(Msg::SettingsZoom.t())
                            .subtitle(Msg::SettingsZoomDesc.t())
                            .trailing(stepper(
                                "zoom",
                                "100%",
                                true,
                                true,
                                Some((StepperPart::Increase, Forced::Rest)),
                            ))
                            .into_any_element(),
                        Row::new(Msg::SettingsCompactThreshold.t())
                            .subtitle(Msg::SettingsCompactThresholdDesc.t())
                            .trailing(stepper(
                                "compact-threshold",
                                "70rem",
                                true,
                                true,
                                Some((StepperPart::Increase, Forced::Rest)),
                            ))
                            .into_any_element(),
                    ]),
                ),
                section(
                    Msg::SettingsGraph.t(),
                    None,
                    None,
                    group([
                        Row::new(Msg::SettingsCompact.t())
                            .subtitle(Msg::SettingsCompactDesc.t())
                            .trailing(switch(
                                "compact-graph",
                                Msg::SettingsCompact.t(),
                                false,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new(Msg::SettingsLaneCompact.t())
                            .subtitle(Msg::SettingsLaneCompactDesc.t())
                            .trailing(switch(
                                "avatar-commit-nodes",
                                Msg::SettingsLaneCompact.t(),
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                    ]),
                ),
                section(
                    Msg::SettingsNetwork.t(),
                    None,
                    None,
                    group([Row::new(Msg::SettingsAutoFetch.t())
                        .subtitle(Msg::SettingsAutoFetchDesc.t())
                        .trailing(switch(
                            "auto-fetch",
                            Msg::SettingsAutoFetch.t(),
                            true,
                            auto_fetch_forced,
                            |_, _, _| {},
                        ))
                        .into_any_element()]),
                ),
            ],
        ),
    )
    .into_any_element()
}
