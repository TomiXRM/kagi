//! Gallery reconstruction for Apple System Settings › Sound (I05/S4).
use gpui::{div, prelude::*, rgb, AnyElement, App, Window};
use gpui_component::{Icon, IconName, Sizable, Size};
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::settings_kit::{
    capsule_button, detail_page, group, icon_tile, ink, popup_value, pt, ring, search_field,
    section, settings_window, sidebar, sidebar_gap, sidebar_item, switch, Forced, Row, Tint,
};

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i05-s4",
    states: &["default", "focus-alert-volume"],
    size: (960., 960.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn apple_sidebar() -> AnyElement {
    let item = |id, icon, label, selected| {
        sidebar_item(
            id,
            icon_tile(icon, Tint::Accent, 20.),
            label,
            None,
            selected,
            Forced::Rest,
        )
        .into_any_element()
    };
    sidebar([
        search_field("sound-search", "Search Settings").into_any_element(),
        item("general", IconName::Settings2, "General", false),
        item("appearance", IconName::Palette, "Appearance", false),
        item(
            "accessibility",
            IconName::CircleUser,
            "Accessibility",
            false,
        ),
        sidebar_gap(),
        item(
            "control-center",
            IconName::LayoutDashboard,
            "Control Center",
            false,
        ),
        item("siri", IconName::Star, "Siri & Spotlight", false),
        item("privacy", IconName::Heart, "Privacy & Security", false),
        sidebar_gap(),
        item("desktop", IconName::Frame, "Desktop & Dock", false),
        item("displays", IconName::Maximize, "Displays", false),
        item("sound", IconName::Bell, "Sound", true),
        item("battery", IconName::Battery, "Battery", false),
        item("lock-screen", IconName::EyeOff, "Lock Screen", false),
        item("users", IconName::User, "Users & Groups", false),
    ])
    .into_any_element()
}

/// Page-local kit gap: a value-positioned slider with optional ticks and
/// independently supplied end content. The same geometry serves all sliders.
fn slider(
    id: &'static str,
    value: f32,
    ticks: Option<usize>,
    leading: AnyElement,
    trailing: AnyElement,
    forced: Forced,
) -> AnyElement {
    let track_w = 242.;
    let thumb_x = 4. + (track_w - 8.) * value;
    let mut track = div()
        .id(id)
        .relative()
        .w(pt(track_w))
        .h(pt(24.))
        .flex()
        .items_center()
        .role(gpui::Role::Slider)
        .aria_label(id)
        .child(
            div()
                .absolute()
                .left(pt(0.))
                .right(pt(0.))
                .top(pt(10.))
                .h(pt(4.))
                .rounded(pt(2.))
                .bg(rgb(ink::control())),
        )
        .child(
            div()
                .absolute()
                .left(pt(0.))
                .top(pt(10.))
                .w(pt(4. + (track_w - 8.) * value))
                .h(pt(4.))
                .rounded(pt(2.))
                .bg(ink::accent()),
        )
        .child(
            div()
                .absolute()
                .left(pt(thumb_x - 5.))
                .top(pt(6.))
                .size(pt(12.))
                .rounded_full()
                .bg(rgb(ink::card()))
                .border_1()
                .border_color(rgb(ink::hairline()))
                .shadow_sm(),
        );
    if let Some(count) = ticks {
        for index in 1..count {
            let x = (track_w - 8.) * index as f32 / count as f32 + 4.;
            track = track.child(
                div()
                    .absolute()
                    .left(pt(x))
                    .top(pt(16.))
                    .w(pt(1.))
                    .h(pt(4.))
                    .bg(ink::secondary()),
            );
        }
    }
    div()
        .flex()
        .items_center()
        .gap(pt(8.))
        .child(leading)
        .child(ring(track, forced == Forced::Focus, 8.))
        .child(trailing)
        .into_any_element()
}

fn sound_symbol(loud: bool) -> AnyElement {
    div()
        .w(pt(13.))
        .text_color(ink::secondary())
        .text_size(pt(if loud { 14. } else { 11. }))
        .child(if loud { "◖" } else { "◔" })
        .into_any_element()
}

fn caption_end(text: &'static str) -> AnyElement {
    div()
        .w(pt(30.))
        .text_size(pt(11.5))
        .text_color(ink::secondary())
        .child(text)
        .into_any_element()
}

/// Page-local kit gap: a labelled checkbox for the output-volume mute state.
fn mute_checkbox() -> AnyElement {
    div()
        .id("mute")
        .flex()
        .items_center()
        .gap(pt(7.))
        .role(gpui::Role::CheckBox)
        .aria_label("Mute")
        .aria_toggled(gpui::Toggled::True)
        .child(
            div()
                .size(pt(14.))
                .rounded(pt(3.))
                .bg(ink::accent())
                .flex()
                .items_center()
                .justify_center()
                .text_color(rgb(ink::window()))
                .child(Icon::new(IconName::Check).with_size(Size::Size(pt(10.)))),
        )
        .child("Mute")
        .into_any_element()
}

/// Page-local kit gap: the kit's `Segmented` is intrinsically sized, while the
/// Sound device chooser requires a two-choice control spanning the group width.
fn full_width_segmented(
    id: &'static str,
    labels: [&'static str; 2],
    selected: usize,
) -> AnyElement {
    let mut control = div()
        .id(id)
        .mx(pt(0.))
        .h(pt(38.))
        .p(pt(3.))
        .flex()
        .rounded(pt(8.))
        .bg(rgb(ink::control()))
        .role(gpui::Role::RadioGroup);
    for (index, label) in labels.into_iter().enumerate() {
        control = control.child(
            div()
                .id((id, index))
                .flex_1()
                .h_full()
                .flex()
                .items_center()
                .justify_center()
                .rounded(pt(6.))
                .role(gpui::Role::RadioButton)
                .aria_label(label)
                .aria_selected(index == selected)
                .when(index == selected, |cell| {
                    cell.bg(ink::selected())
                        .font_weight(gpui::FontWeight::MEDIUM)
                        .shadow_sm()
                })
                .when(index != selected, |cell| cell.text_color(ink::secondary()))
                .child(label),
        );
    }
    control.into_any_element()
}

/// Page-local kit gap: a fixed device-choice table with one selected row.
fn device_table() -> AnyElement {
    let devices = [
        ("DELL U2723QE", "DisplayPort", false),
        ("Shure MV7", "USB", false),
        ("MacBook Pro Speakers", "Built-in", true),
        ("AirPods Pro", "Bluetooth", false),
        ("AirMac Express", "AirPlay", false),
        ("tomixrm-m1", "AirPlay", false),
    ];
    let mut table = div().flex().flex_col().py(pt(5.));
    table = table.child(
        div()
            .h(pt(24.))
            .px(pt(8.))
            .flex()
            .items_center()
            .text_size(pt(11.5))
            .text_color(ink::secondary())
            .child(div().flex_1().child("Name"))
            .child(div().w(pt(112.)).child("Type")),
    );
    for (name, kind, selected) in devices {
        table = table.child(
            div()
                .h(pt(28.))
                .px(pt(8.))
                .rounded(pt(6.))
                .flex()
                .items_center()
                .when(selected, |row| {
                    row.bg(ink::selected())
                        .font_weight(gpui::FontWeight::MEDIUM)
                })
                .child(div().flex_1().min_w_0().truncate().child(name))
                .child(div().w(pt(112.)).text_color(ink::secondary()).child(kind)),
        );
    }
    table.into_any_element()
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let alert_volume_state = if cx.state == "focus-alert-volume" {
        Forced::Focus
    } else {
        Forced::Rest
    };
    let alert_controls = div()
        .flex()
        .items_center()
        .gap(pt(8.))
        .child(popup_value("alert-sound", "Boop", Forced::Rest))
        .child(capsule_button("play-alert", "Play", Forced::Rest));

    let sound_effects = section(
        "Sound Effects",
        None,
        None,
        group([
            Row::new("Alert sound")
                .trailing(alert_controls)
                .into_any_element(),
            Row::new("Play sound effects through")
                .trailing(popup_value(
                    "sound-effects-output",
                    "MacBook Pro Speakers",
                    Forced::Rest,
                ))
                .into_any_element(),
            Row::new("Alert volume")
                .trailing(slider(
                    "alert-volume",
                    0.88,
                    Some(9),
                    sound_symbol(false),
                    sound_symbol(true),
                    alert_volume_state,
                ))
                .into_any_element(),
            Row::new("Play sound on startup")
                .trailing(switch(
                    "startup",
                    "Play sound on startup",
                    true,
                    Forced::Rest,
                    |_, _, _| {},
                ))
                .into_any_element(),
            Row::new("Play user interface sound effects")
                .trailing(switch(
                    "ui-effects",
                    "Play user interface sound effects",
                    true,
                    Forced::Rest,
                    |_, _, _| {},
                ))
                .into_any_element(),
            Row::new("Play feedback when volume is changed")
                .trailing(switch(
                    "volume-feedback",
                    "Play feedback when volume is changed",
                    false,
                    Forced::Rest,
                    |_, _, _| {},
                ))
                .into_any_element(),
        ]),
    );

    let output_input = section(
        "Output & Input",
        None,
        None,
        group([
            full_width_segmented("output-input", ["Output", "Input"], 0),
            device_table(),
        ]),
    );

    let output_controls = group([
        Row::new("Output volume")
            .trailing(slider(
                "output-volume",
                0.16,
                None,
                sound_symbol(false),
                sound_symbol(true),
                Forced::Rest,
            ))
            .into_any_element(),
        Row::new("").trailing(mute_checkbox()).into_any_element(),
        Row::new("Balance")
            .trailing(slider(
                "balance",
                0.5,
                None,
                caption_end("Left"),
                caption_end("Right"),
                Forced::Rest,
            ))
            .into_any_element(),
    ])
    .into_any_element();

    settings_window(
        apple_sidebar(),
        div()
            .flex_1()
            .min_w_0()
            .relative()
            .child(detail_page(
                "sound-scroll",
                "Sound",
                [sound_effects, output_input, output_controls],
            ))
            .child(
                div()
                    .absolute()
                    .right(pt(22.))
                    .bottom(pt(16.))
                    .child(capsule_button("help", "?", Forced::Rest)),
            ),
    )
    .into_any_element()
}
