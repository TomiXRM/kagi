//! iCloud settings composed from the Kagi settings kit.
use gpui::{div, prelude::*, relative, rgb, AnyElement, App, Div, FontWeight, Window};
use gpui_component::IconName;
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::settings_kit::*;

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i05-s5",
    states: &["default", "focus-card", "hover-manage", "focus-nav"],
    size: (1000., 1040.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn words(cx: &GalleryContext, en: &'static str, ja: &'static str) -> &'static str {
    if cx.lang == kagi::ui::i18n::Lang::Ja {
        ja
    } else {
        en
    }
}

fn forced(cx: &GalleryContext, name: &str, state: Forced) -> Forced {
    if cx.state == name {
        state
    } else {
        Forced::Rest
    }
}

/// Kit gap: a determinate progress track, with its fill inside the track.
fn progress_bar(fraction: f32, tint: Tint) -> Div {
    div()
        .w_full()
        .h(pt(8.))
        .rounded(pt(4.))
        .overflow_hidden()
        .bg(rgb(ink::control()))
        .child(
            div()
                .h_full()
                .w(relative(fraction.clamp(0., 1.)))
                .rounded(pt(4.))
                .bg(rgb(tint.color())),
        )
}

/// A card composes the kit's group, row and tile; only its grid allocation is local.
fn cloud_card(
    symbol: IconName,
    title: &'static str,
    subtitle: &'static str,
    state: Forced,
) -> AnyElement {
    let body = group([Row::new(title)
        .subtitle(subtitle)
        .leading(icon_tile(symbol, Tint::Accent, 26.))
        .into_any_element()])
    .h(pt(102.))
    .justify_center();
    ring(
        div().w_full().child(body),
        state == Forced::Focus,
        metrics::GROUP_RADIUS,
    )
    .when(state == Forced::Hover, |el| {
        el.bg(rgb(ink::control_hover()))
    })
    .when(state == Forced::Pressed, |el| {
        el.bg(rgb(ink::control_pressed()))
    })
    .when(state == Forced::Disabled, |el| {
        el.opacity(ink::disabled_opacity())
    })
    .into_any_element()
}

fn card_grid(cards: [(IconName, &'static str, &'static str); 6], first: Forced) -> Div {
    let mut rows = Vec::new();
    let mut cards = cards.into_iter().enumerate();
    for _ in 0..2 {
        rows.push(
            div()
                .w_full()
                .flex()
                .gap(pt(8.))
                .children(
                    cards
                        .by_ref()
                        .take(3)
                        .map(|(index, (icon, title, subtitle))| {
                            div().flex_1().min_w_0().child(cloud_card(
                                icon,
                                title,
                                subtitle,
                                if index == 0 { first } else { Forced::Rest },
                            ))
                        }),
                ),
        );
    }
    div().flex().flex_col().gap(pt(8.)).children(rows)
}

/// Kit gap: a circular account badge containing a kit-owned symbol tile.
fn account_badge() -> Div {
    div()
        .size(pt(52.))
        .flex_shrink_0()
        .flex()
        .items_center()
        .justify_center()
        .rounded(pt(26.))
        .border_1()
        .border_color(rgb(ink::hairline()))
        .bg(rgb(ink::card()))
        .child(icon_tile(IconName::Inbox, Tint::Accent, 32.))
}
