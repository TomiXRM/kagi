    name: "lab-i06-s2",
    states: &[
        "default",
        "hover-choice",
        "pressed-choice",
        "focus-choice",
        "disabled-choice",
        "focus-switch",
        "focus-slider",

fn text(cx: &GalleryContext, en: &'static str, ja: &'static str) -> &'static str {
    if cx.lang.slug() == "ja" {
        ja
    } else {
        en
    }
}

fn forced(cx: &GalleryContext, part: &str) -> Forced {
    cx.state
        .strip_suffix(part)
        .and_then(|s| s.strip_suffix('-'))
        .and_then(Forced::parse)
        .unwrap_or(Forced::Rest)
}

/// Kit gap: a labelled visual choice. Selection surrounds its own preview;
/// interaction changes only the fill/ring, never the selected value.
fn choice_tile(
    id: &'static str,
    preview: AnyElement,
    label: &'static str,
    selected: bool,
    forced: Forced,
) -> AnyElement {
    let fill = match forced {
        Forced::Hover => ink::control_hover(),
        Forced::Pressed => ink::control_pressed(),
        _ => ink::control(),
    };
    div()
        .id(id)
        .w_full()
        .min_w_0()
        .flex_1()
        .flex()
        .flex_col()
        .gap(pt(5.))
        .items_center()
        .role(Role::RadioButton)
        .aria_label(label)
        .aria_selected(selected)
        .when(forced == Forced::Disabled, |el| {
            el.opacity(ink::disabled_opacity())
        })
        .child(ring(
            div()
                .w_full()
                .min_w_0()
                .h(pt(80.))
                .p(pt(4.))
                .bg(rgb(fill))
                .child(preview),
            selected || forced == Forced::Focus,
            metrics::GROUP_RADIUS,
        ))
        .child(
            div()
                .text_size(pt(metrics::SECONDARY))
                .text_color(if selected {
                    ink::text()
                } else {
                    ink::secondary()
                })
                .when(selected, |el| el.font_weight(FontWeight::SEMIBOLD))
                .child(label),
        )
        .into_any_element()
}

/// A miniature window preview, rather than an icon segment. All five scenes
/// share the same canvas and category hue, with theme-derived surface inks.
fn preview(symbol: IconName, glass: Option<bool>) -> AnyElement {
    div()
        .size_full()
        .flex()
        .items_center()
        .justify_center()
        .rounded(pt(metrics::CONTROL_RADIUS))
        .bg(rgb(Tint::Hue(Hue::Blue).color()))
        .child(if let Some(glass) = glass {
            div()
                .size(pt(40.))
                .rounded_full()
                .border_1()
                .border_color(ink::secondary())
                .bg(rgb(if glass {
                    ink::control_hover()
                } else {
                    ink::window()
                }))
                .into_any_element()
        } else {
            div()
                .w(pt(48.))
                .h(pt(43.))
                .p(pt(4.))
                .flex()
                .flex_col()
                .gap(pt(3.))
                .rounded(pt(metrics::CONTROL_RADIUS))
                .bg(rgb(ink::card()))
                .border_1()
                .border_color(ink::secondary())
                .child(div().flex().gap(pt(2.)).children(
                    (0..3).map(|_| div().size(pt(3.)).rounded_full().bg(ink::secondary())),
                ))
                .child(
                    div()
                        .flex_1()
                        .w_full()
                        .flex()
                        .items_center()
                        .justify_center()
                        .rounded(pt(3.))
                        .bg(rgb(ink::control()))
                        .child(icon_tile(symbol, Tint::Hue(Hue::Indigo), 18.)),
                )
                .into_any_element()
        })
        .into_any_element()
}

fn appearance(cx: &GalleryContext) -> AnyElement {
    let cluster = |title, choices: Vec<AnyElement>, weight| {
        div()
            .w_full()
            .min_w_0()
            .flex_grow(weight)
            .flex_basis(pt(0.))
            .flex()
            .flex_col()
            .gap(pt(8.))
            .child(caption(title))
            .child(
                div()
                    .w_full()
                    .min_w_0()
                    .flex()
                    .gap(pt(8.))
                    .children(choices),
            )
    };
    section(
        text(cx, "Appearance", "外観"),
        None,
        None,
        group([
            div()
                .w_full()
                .min_w_0()
                .py(pt(12.))
                .flex()
                .gap(pt(12.))
                .child(cluster(
                    text(cx, "Color scheme", "カラースキーム"),
                    vec![
                        choice_tile(
                            "auto",
                            preview(IconName::ALargeSmall, None),
                            text(cx, "Auto", "自動"),
                            true,
                            Forced::Rest,
                        ),
                        choice_tile(
                            "light",
                            preview(IconName::Sun, None),
                            text(cx, "Light", "ライト"),
                            false,
                            forced(cx, "choice"),
                        ),
                        choice_tile(
                            "dark",
                            preview(IconName::Moon, None),
                            text(cx, "Dark", "ダーク"),
                            false,
                            Forced::Rest,
                        ),
                    ],
                    3.,
                ))
                .child(vertical_divider())
                .child(cluster(
                    text(cx, "Interface style", "インターフェイススタイル"),
                    vec![
                        choice_tile(
                            "glass",
                            preview(IconName::CircleUser, Some(true)),
                            text(cx, "Glass", "ガラス"),
                            false,
                            Forced::Rest,
                        ),
                        choice_tile(
                            "classic",
                            preview(IconName::CircleUser, Some(false)),
                            text(cx, "Classic", "クラシック"),
                            true,
                            Forced::Rest,
                        ),
                    ],
                    2.,
                ))
                .into_any_element(),
            Row::new(text(cx, "Tint with theme", "テーマの色を使用"))
                .trailing(switch(
                    "tint",
                    text(cx, "Tint with theme", "テーマの色を使用"),
                    true,
                    forced(cx, "switch"),
                    |_, _, _| {},
                ))
                .into_any_element(),
        ]),
    )
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    settings_window(
        rex_sidebar(cx),
        detail_page(
            "interface-scroll",
            text(cx, "Interface", "インターフェイス"),
            [
                appearance(cx),
                section(
                    text(cx, "Tabs", "タブ"),
                    None,
                    None,
                    group([
                        Row::new(text(cx, "Tab style", "タブのスタイル"))
                            .trailing(Segmented::new(
                                "tab-style",
                                [
                                    text(cx, "Above", "上部"),
                                    text(cx, "Left Side", "左側"),
                                    text(cx, "Right Side", "右側"),
                                ],
                                0,
                            ))
                            .into_any_element(),
                        Row::new(text(cx, "Fill available space", "利用可能な領域を埋める"))
                            .trailing(switch(
                                "fill",
                                text(cx, "Fill available space", "利用可能な領域を埋める"),
                                false,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                        Row::new(text(cx, "Show tab icons", "タブアイコンを表示"))
                            .trailing(
                                div()
                                    .flex()
                                    .items_center()
                                    .gap(pt(8.))
                                    .child(
                                        Slider::new("tab-icons", 0.5)
                                            .width(150.)
                                            .ticks(5)
                                            .forced(forced(cx, "slider")),
                                    )
                                    .child(caption("3")),
                            )
                            .into_any_element(),
                        Row::new(text(
                            cx,
                            "Use color for tab icons",
                            "タブアイコンに色を使用",
                        ))
                        .trailing(popup_value(
                            "icon-color",
                            text(cx, "Always", "常に"),
                            Forced::Rest,
                        ))
                        .into_any_element(),
                        Row::new(text(cx, "Shell icon theme", "シェルアイコンのテーマ"))
                            .trailing(popup_value(
                                "shell-theme",
                                text(cx, "Default", "デフォルト"),
                                Forced::Rest,
                            ))
                            .into_any_element(),
                    ]),
                ),
                section(
                    text(cx, "Panes", "ペイン"),
                    None,
                    None,
                    group([
                        Row::new(text(cx, "Pane style", "ペインのスタイル"))
                            .trailing(Segmented::new(
                                "pane-style",
                                [
                                    text(cx, "Comfortable", "ゆったり"),
                                    text(cx, "Compact", "コンパクト"),
                                ],
                                0,
                            ))
                            .into_any_element(),
                        Row::new(text(cx, "Show pane headers", "ペインのヘッダーを表示"))
                            .trailing(switch(
                                "headers",
                                text(cx, "Show pane headers", "ペインのヘッダーを表示"),
                                true,
                                Forced::Rest,
                                |_, _, _| {},
                            ))
                            .into_any_element(),
                    ]),
                ),
            ],
        ),
    )
    .into_any_element()
}
