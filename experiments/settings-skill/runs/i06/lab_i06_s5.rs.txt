/// Page-local composition for a three-column cloud-service summary card.
/// The kit has no card-grid part; its tile remains the kit-owned symbol tile.
fn cloud_card(
    id: &'static str,
    icon: IconName,
    hue: Hue,
    title: &'static str,
    subtitle: &'static str,
    forced: Forced,
) -> AnyElement {
    let card = div()
        .id(id)
        .flex_1()
        .min_w_0()
        .h(pt(92.))
        .p(pt(12.))
        .flex()
        .items_start()
        .gap(pt(10.))
        .rounded(pt(9.))
        .bg(rgb(ink::control()))
        .when(forced == Forced::Hover, |el| {
            el.bg(rgb(ink::control_hover()))
        })
        .when(forced == Forced::Pressed, |el| {
            el.bg(rgb(ink::control_pressed()))
        })
        .when(forced == Forced::Disabled, |el| {
            el.opacity(ink::disabled_opacity())
        })
        .child(icon_tile(icon, Tint::Hue(hue), 30.))
        .child(
            div()
                .min_w_0()
                .flex_1()
                .flex()
                .flex_col()
                .gap(pt(3.))
                .child(
                    div()
                        .min_w_0()
                        .font_weight(FontWeight::SEMIBOLD)
                        .line_clamp(2)
                        .child(title),
                )
                .child(
                    div()
                        .min_w_0()
                        .text_size(pt(11.5))
                        .text_color(ink::secondary())
                        .line_clamp(2)
                        .child(subtitle),
                ),
        );
    ring(card, forced == Forced::Focus, 9.).into_any_element()
}

/// Page-local full-width storage meter: the kit owns the actual progress bar.
fn storage_meter(fraction: f32, tint: Tint) -> AnyElement {
    div()
        .w_full()
        .pt_0()
        .child(kagi::ui::settings_kit::progress_bar(fraction, tint))
        .into_any_element()
}

/// Page-local Help button because the kit intentionally has no Help control.
fn help_button(forced: Forced) -> AnyElement {
    let fill = match forced {
        Forced::Hover => ink::control_hover(),
        Forced::Pressed => ink::control_pressed(),
        _ => ink::control(),
    };
    ring(
        div()
            .id("help")
            .size(pt(28.))
            .rounded_full()
            .role(Role::Button)
            .aria_label("Help")
            .flex()
            .items_center()
            .justify_center()
            .bg(rgb(fill))
            .font_weight(FontWeight::BOLD)
            .child("?")
            .hover(|el| el.bg(rgb(ink::control_hover())))
            .active(|el| el.bg(rgb(ink::control_pressed())))
            .when(forced == Forced::Disabled, |el| {
                el.opacity(ink::disabled_opacity())
            }),
        forced == Forced::Focus,
        14.,
    )
    .into_any_element()
}

fn card_grid(cards: [AnyElement; 6]) -> AnyElement {
    let [a, b, c, d, e, f] = cards;
    div()
        .flex()
        .flex_col()
        .gap(pt(10.))
        .child(
            div()
                .w_full()
                .flex()
                .gap(pt(10.))
                .child(a)
                .child(b)
                .child(c),
        )
        .child(
            div()
                .w_full()
                .flex()
                .gap(pt(10.))
                .child(d)
                .child(e)
                .child(f),
        )
        .into_any_element()
}

                        .navigation("advanced", Forced::Rest)
                        .into_any_element(),
                    Row::new("Access iCloud Data on the Web")
                        .trailing(div().text_color(ink::secondary()).child("On"))
                        .navigation("web-data", Forced::Rest)
