//! S8 — a settings information architecture built entirely from settings_kit.
use gpui::{div, prelude::*, AnyElement, App, Role, Window};
use gpui_component::IconName;
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::i18n::Msg;
use kagi::ui::settings_kit::{
    capsule_button, detail_page, group, icon_tile, ink, popup_value, pt, ring, search_field,
    section, settings_window, sidebar, sidebar_gap, sidebar_item, stepper, switch, Forced, Hue,
    Row, Segmented, StepperPart, Tint,
};

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i07-s8",
    states: &[
        "default",
        "pane-general",
        "pane-graph",
        "pane-network-sync",
        "pane-behaviour",
        "pane-language",
        "pane-smart-commit",
        "pane-analyze",
    ],
    size: (900., 692.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Pane {
    General,
    Graph,
    NetworkSync,
    Behaviour,
    Language,
    SmartCommit,
    Analyze,
}

impl Pane {
    fn from_state(state: &str) -> Self {
        match state {
            "pane-graph" => Self::Graph,
            "pane-network-sync" => Self::NetworkSync,
            "pane-behaviour" => Self::Behaviour,
            "pane-language" => Self::Language,
            "pane-smart-commit" => Self::SmartCommit,
            "pane-analyze" => Self::Analyze,
            _ => Self::General,
        }
    }
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let pane = Pane::from_state(cx.state);
    settings_window(settings_sidebar(pane), pane_detail(pane)).into_any_element()
}

fn settings_sidebar(selected: Pane) -> AnyElement {
    let item = |id, icon, hue, label, pane| {
        sidebar_item(
            id,
            icon_tile(icon, Tint::Hue(hue), 20.),
            label,
            None,
            selected == pane,
            Forced::Rest,
        )
        .into_any_element()
    };
    sidebar([
        search_field("settings-search", "Search Settings").into_any_element(),
        item(
            "general",
            IconName::Settings2,
            Hue::Gray,
            "General",
            Pane::General,
        ),
        item(
            "graph",
            IconName::ChartPie,
            Hue::Orange,
            "Graph",
            Pane::Graph,
        ),
        item(
            "network-sync",
            IconName::Globe,
            Hue::Blue,
            "Network & Sync",
            Pane::NetworkSync,
        ),
        item(
            "behaviour",
            IconName::SquareTerminal,
            Hue::Gray,
            "Behaviour",
            Pane::Behaviour,
        ),
        sidebar_gap(),
        item(
            "language",
            IconName::ALargeSmall,
            Hue::Indigo,
            Msg::SettingsLanguage.t(),
            Pane::Language,
        ),
        item(
            "smart-commit",
            IconName::Bot,
            Hue::Purple,
            Msg::SettingsSmartCommit.t(),
            Pane::SmartCommit,
        ),
        item(
            "analyze",
            IconName::Inspector,
            Hue::Teal,
            Msg::SettingsAnalyze.t(),
            Pane::Analyze,
        ),
    ])
    .into_any_element()
}

fn pane_detail(pane: Pane) -> AnyElement {
    match pane {
        Pane::General => general_pane(),
        Pane::Graph => graph_pane(),
        Pane::NetworkSync => network_pane(),
        Pane::Behaviour => behaviour_pane(),
        Pane::Language => language_pane(),
        Pane::SmartCommit => smart_commit_pane(),
        Pane::Analyze => analyze_pane(),
    }
}

/// Page-local kit gap: the kit intentionally has no multiline text editor.
fn analyze_ignore_editor(id: &'static str, value: &str, forced: Forced) -> impl IntoElement {
    let editor = div()
        .id(id)
        .w_full()
        .min_w_0()
        .h(pt(82.))
        .p(pt(10.))
        .flex()
        .flex_col()
        .gap(pt(3.))
        .rounded(pt(7.))
        .bg(gpui::rgb(ink::control()))
        .text_color(ink::secondary())
        .text_size(pt(11.5))
        .role(Role::TextInput)
        .aria_label(Msg::A11yAnalyzeIgnorePatterns.t())
        .children(value.lines().map(|line| div().child(line.to_owned())));
    ring(editor, forced == Forced::Focus, 7.)
}
