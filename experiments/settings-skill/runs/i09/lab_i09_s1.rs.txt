//! Rex Notifications, composed from the Kagi settings kit.
use gpui::{div, prelude::*, rgb, AnyElement, App, ElementId, Role, Window};
use gpui_component::{Icon, IconName, Sizable as _, Size};
use kagi::ui::gallery_host::{self, GalleryContext, GalleryPage};
use kagi::ui::settings_kit::*;

static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i09-s1",
    states: &[
        "default",
        "rest-picker",
        "selected-bottom-right",
        "selected-top-left",
        "selected-top-right",
        "selected-bottom-left",
        "focus-picker",
        "hover-picker",
        "focus-switch",
    ],
    size: (900., 740.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

const POSITIONS: [&str; 6] = [
    "top left",
    "top center",
    "top right",
    "bottom left",
    "bottom center",
    "bottom right",
];

/// One row-major value-to-position mapping for cells and their indicators.
fn position_origin(cell: usize) -> (f32, f32) {
    (2. + (cell % 3) as f32 * 34., 1. + (cell / 3) as f32 * 10.)
}

/// Kit gap: a six-position visual choice, kept at CONTROL_HEIGHT beside rows.
/// The selected check is a child of its cell, so it shares the cell's mapping.
fn position_picker(selected_cell: usize, forced: Forced) -> impl IntoElement {
    let selected = selected_cell.min(POSITIONS.len() - 1);
    let fill = match forced {
        Forced::Hover => ink::control_hover(),
        Forced::Pressed => ink::control_pressed(),
        _ => ink::control(),
    };
    let track = div()
        .id("notification-position")
        .relative()
        .w(pt(104.))
        .h(pt(metrics::CONTROL_HEIGHT))
        .flex_shrink_0()
        .bg(rgb(fill))
        .role(Role::RadioGroup)
        .aria_label("Notification position")
        .children(POSITIONS.into_iter().enumerate().map(|(index, label)| {
            let (x, y) = position_origin(index);
            div()
                .id(ElementId::NamedInteger(
                    "notification-cell".into(),
                    index as u64,
                ))
                .absolute()
                .left(pt(x))
                .top(pt(y))
                .w(pt(28.))
                .h(pt(8.))
                .rounded(pt(4.))
                .flex()
                .items_center()
                .justify_center()
                .role(Role::RadioButton)
                .aria_label(label)
                .aria_selected(index == selected)
                .when(index != selected, |el| el.bg(rgb(ink::card())))
                .when(index == selected, |el| {
                    el.bg(ink::accent())
                        .text_color(rgb(ink::window()))
                        .child(Icon::new(IconName::Check).with_size(Size::Size(pt(8.))))
                })
        }));
    ring(track, forced == Forced::Focus, metrics::CONTROL_RADIUS)
        .when(forced == Forced::Disabled, |el| {
            el.opacity(ink::disabled_opacity())
        })
}

fn rex_sidebar() -> AnyElement {
    let item = |id: &'static str, icon, hue, label: &'static str| {
        sidebar_item(
            id,
            icon_tile(icon, Tint::Hue(hue), metrics::TILE),
            label,
            None,
            id == "notifications",
            Forced::Rest,
        )
        .into_any_element()
    };
    sidebar([
        search_field("search", "Search Settings").into_any_element(),
        sidebar_item(
            "server",
            icon_tile(IconName::Bot, Tint::Hue(Hue::Purple), metrics::TILE_LARGE),
            "Server",
            Some("Running".into()),
            false,
            Forced::Rest,
        )
        .into_any_element(),
        sidebar_gap(),
        item(
            "interface",
            IconName::LayoutDashboard,
            Hue::Blue,
            "Interface",
        ),
        item("themes", IconName::Palette, Hue::Indigo, "Themes"),
        sidebar_gap(),
        item("terminal", IconName::SquareTerminal, Hue::Gray, "Terminal"),
        item("shortcuts", IconName::CaseSensitive, Hue::Gray, "Shortcuts"),
        item(
            "integrations",
            IconName::Github,
            Hue::Orange,
            "Integrations",
        ),
        item("notifications", IconName::Bell, Hue::Red, "Notifications"),
        item("remote-hosts", IconName::Globe, Hue::Blue, "Remote Hosts"),
        sidebar_gap(),
        item("developer", IconName::Settings2, Hue::Gray, "Developer"),
    ])
    .into_any_element()
}

fn toggle_row(id: &'static str, title: &'static str, checked: bool, forced: Forced) -> AnyElement {
    Row::new(title)
        .trailing(switch(id, title, checked, forced, |_, _, _| {}))
        .into_any_element()
}

fn activity_row(
    id: &'static str,
    title: &'static str,
    subtitle: &'static str,
    icon: IconName,
    tint: Tint,
) -> AnyElement {
    Row::new(title)
        .subtitle(subtitle)
        .leading(icon_tile(icon, tint, 26.))
        .navigation(id, Forced::Rest)
        .into_any_element()
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let selected = match cx.state {
        "selected-top-left" => 0,
        "selected-top-right" => 2,
        "selected-bottom-left" => 3,
        _ => 5,
    };
    let picker_state = match cx.state {
        "focus-picker" => Forced::Focus,
        "hover-picker" => Forced::Hover,
        _ => Forced::Rest,
    };
    let switch_state = if cx.state == "focus-switch" {
        Forced::Focus
    } else {
        Forced::Rest
    };
    settings_window(
        rex_sidebar(),
        detail_page(
            "notifications-scroll",
            "Notifications",
            [
                section(
                    "In-App Notifications",
                    None,
                    None,
                    group([Row::new("Notification position")
                        .subtitle(format!("Displayed {}", POSITIONS[selected]))
                        .trailing(position_picker(selected, picker_state))
                        .into_any_element()]),
                ),
                section(
                    "Program Activity",
                    Some("How status from processes and agents is handled".into()),
                    None,
                    div()
                        .w_full()
                        .min_w_0()
                        .flex()
                        .flex_col()
                        .gap(pt(10.))
                        .child(group([
                            toggle_row(
                                "activity-tabs",
                                "Show activity in tabs",
                                true,
                                switch_state,
                            ),
                            toggle_row(
                                "system-notifications",
                                "Show system notifications",
                                true,
                                Forced::Rest,
                            ),
                            toggle_row("dock-badge", "Badge Dock icon", true, Forced::Rest),
                            toggle_row("sounds", "Play sounds", true, Forced::Rest),
                        ]))
                        .child(group([
                            activity_row(
                                "working",
                                "Working",
                                "Tabs",
                                IconName::Ellipsis,
                                Tint::Hue(Hue::Blue),
                            ),
                            activity_row(
                                "blocked",
                                "Blocked",
                                "Tabs, Notifications, Dock Badge, Sound",
                                IconName::CircleX,
                                Tint::Warning,
                            ),
                            activity_row(
                                "error",
                                "Error",
                                "Tabs, Notifications, Dock Badge, Sound",
                                IconName::TriangleAlert,
                                Tint::Danger,
                            ),
                            activity_row(
                                "done",
                                "Done",
                                "Tabs, Notifications, Dock Badge, Sound",
                                IconName::CircleCheck,
                                Tint::Success,
                            ),
                        ]))
                        .child(group([
                            toggle_row(
                                "idle",
                                "Include idle processes in activity",
                                true,
                                Forced::Rest,
                            ),
                            toggle_row(
                                "each-process",
                                "Show each process in tabs",
                                false,
                                Forced::Rest,
                            ),
                        ])),
                ),
            ],
        ),
    )
    .into_any_element()
}
