/// Kit gap: a round decorative account badge, containing a kit symbol tile.
fn cloud_badge() -> Div {

        .size(pt(52.))

/// Kit gap: reusable destination card composition from group, row and tile.
/// Fixed outer height reserves space for a wrapped title above the subtitle.
fn cloud_card(
    id: &'static str,
    symbol: IconName,
    hue: Hue,
    title: &'static str,
    subtitle: &'static str,
    forced: Forced,
) -> AnyElement {
    let card = group([Row::new(title)
        .subtitle(subtitle)
        .leading(icon_tile(symbol, Tint::Hue(hue), 26.))
        .into_any_element()])
    .id(id)
    .w_full()
    .min_w_0()
    .h(pt(104.))
    .justify_center()
    .bg(rgb(ink::control()))
    .when(forced == Forced::Hover, |el| {
        el.bg(rgb(ink::control_hover()))
    })
    .when(forced == Forced::Pressed, |el| {
        el.bg(rgb(ink::control_pressed()))
    })
    .when(forced == Forced::Disabled, |el| {
        el.opacity(ink::disabled_opacity())
    });
    let card = ring(card, forced == Forced::Focus, metrics::GROUP_RADIUS)
        .when(forced != Forced::Focus, |el| {
            el.border(pt(1.)).border_color(rgb(ink::hairline()))
        });
    div().flex_1().min_w_0().child(card).into_any_element()
}

fn card_grid(cards: [AnyElement; 6]) -> Div {
    let mut cards = cards.into_iter();
    let rows = (0..2)
        .map(|_| {
            div()
                .w_full()
                .min_w_0()
                .flex()
                .gap(pt(8.))
                .children(cards.by_ref().take(3).collect::<Vec<_>>())
        })
        .collect::<Vec<_>>();
    group([div()
        .w_full()
        .min_w_0()
        .py(pt(12.))
        .flex()
        .flex_col()
        .gap(pt(8.))
        .children(rows)
        .into_any_element()])
    .w_full()
    .min_w_0()
}

/// Kit gap: the section API has no leading accessory. Keep its real heading
/// and trailing accessory, with a status dot immediately before the heading.
fn saved_section(ja: bool, cards: [AnyElement; 6]) -> AnyElement {
    div()
        .relative()
        .w_full()
        .min_w_0()
        .child(section(
            text(ja, "Saved to iCloud", "iCloudに保存済み"),
            None,
            Some(
                capsule_button("see-all", text(ja, "See All", "すべて見る"), Forced::Rest)
                    .into_any_element(),
            ),
            card_grid(cards),
        ))
        .child(
            div()
                .absolute()
                .left_0()
                .top(pt(11.))
                .size(pt(6.))
                .rounded_full()
                .bg(rgb(Tint::Success.color())),
        )
        .into_any_element()
}
