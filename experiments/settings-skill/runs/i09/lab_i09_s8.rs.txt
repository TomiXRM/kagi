fn sidebar_for(selected: &str) -> AnyElement {
    let item = |id: &'static str, icon, hue, label: &'static str| {
        sidebar_item(
            id,
            icon_tile(icon, Tint::Hue(hue), 20.),
            label,
            None,
            selected == id,
            Forced::Rest,
        )
        .into_any_element()
    };

    sidebar([
        search_field("search", "Search Settings").into_any_element(),
        item(
            "appearance",
            IconName::Palette,
            Hue::Indigo,
            Msg::SettingsAppearance.t(),
        ),
        item("graph", IconName::Map, Hue::Orange, Msg::SettingsGraph.t()),
        item("editor", IconName::File, Hue::Teal, "Editor"),
        item(
            "network",
            IconName::Globe,
            Hue::Blue,
            Msg::SettingsNetwork.t(),
        ),
        sidebar_gap(),
        item("terminal", IconName::SquareTerminal, Hue::Gray, "Terminal"),
        item(
            "smart-commit",
            IconName::Bot,
            Hue::Purple,
            Msg::SettingsSmartCommit.t(),
        ),
        item(
            "analyze",
            IconName::ChartPie,
            Hue::Teal,
            Msg::SettingsAnalyze.t(),
        ),
        sidebar_gap(),
        item(
            "language",
            IconName::ALargeSmall,
            Hue::Blue,
            Msg::SettingsLanguage.t(),
        ),
    ])
    .into_any_element()
}

/// Page-local kit gap: the kit has no multi-line text editor. Its focus ring,
/// colours, radius, and scale all use kit tokens; the example is display-only.
fn multiline_patterns(value: impl Into<gpui::SharedString>, forced: Forced) -> AnyElement {
    let value = value.into();
    let field = div()
        .id("analyze-patterns")
        .w_full()
        .h(pt(104.))
        .p(pt(10.))
        .rounded(pt(metrics::CONTROL_RADIUS))
        .bg(rgb(ink::control()))
        .text_color(ink::text())
        .text_size(pt(metrics::BODY))
        .child(value);
    ring(field, forced == Forced::Focus, metrics::CONTROL_RADIUS).into_any_element()
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let selected = match cx.state {
        "pane-graph" => "graph",
        "pane-editor" => "editor",
        "pane-network" => "network",
        "pane-terminal" => "terminal",
        "pane-smart-commit" => "smart-commit",
        "pane-analyze" => "analyze",
        "pane-language" => "language",
        _ => "appearance",
    };
    let (title, body) = match selected {
        "graph" => (Msg::SettingsGraph.t(), graph()),
        "editor" => ("Editor", editor()),
        "network" => (Msg::SettingsNetwork.t(), network()),
        "terminal" => ("Terminal", terminal()),
        "smart-commit" => (Msg::SettingsSmartCommit.t(), smart_commit()),
        "analyze" => (Msg::SettingsAnalyze.t(), analyze()),
        "language" => (Msg::SettingsLanguage.t(), language()),
        _ => (Msg::SettingsAppearance.t(), appearance()),
    };
    let sections = if selected == "appearance" {
        vec![body, custom_themes()]
    } else {
        vec![body]
    };

    settings_window(
        sidebar_for(selected),
        detail_page("settings-detail", title, sections),
    )
    .into_any_element()
}
