/// The missing settings-kit control: a six-cell notification position chooser.
/// `selected` is `(row, column)`, with values in `0..2` and `0..3` respectively.
fn position_picker(selected: (usize, usize), forced: Forced) -> AnyElement {
    let mut picker = div()
        .id("notification-position")
        .w(pt(114.))
        .p(pt(4.))
        .flex()
        .flex_col()
        .gap(pt(4.))
        .rounded(pt(metrics::CONTROL_RADIUS))
        .bg(rgb(ink::control()))
        .role(Role::RadioGroup)
        .aria_label("Notification position");

    for row in 0..2 {
        let mut line = div().w_full().flex().gap(pt(4.));
        for column in 0..3 {
            let is_selected = selected == (row, column);
            let cell = div()
                .id(format!("notification-position-{row}-{column}"))
                .h(pt(metrics::CONTROL_HEIGHT))
                .flex_1()
                .min_w_0()
                .flex()
                .items_center()
                .justify_center()
                .rounded(pt(metrics::CONTROL_RADIUS - 2.))
                .role(Role::RadioButton)
                .aria_label(format!("Position row {} column {}", row + 1, column + 1))
                .aria_selected(is_selected)
                .when(is_selected, |cell| {
                    cell.bg(ink::selected()).child(
                        Icon::new(IconName::Check)
                            .with_size(Size::Size(pt(13.)))
                            .text_color(ink::accent_text()),
                    )
                })
                .when(!is_selected, |cell| cell.bg(rgb(ink::card())));
            line = line.child(ring(
                cell,
                forced == Forced::Focus && is_selected,
                metrics::CONTROL_RADIUS - 2.,
            ));
        }
        picker = picker.child(line);
    }

    ring(picker, forced == Forced::Focus, metrics::CONTROL_RADIUS).into_any_element()
}
