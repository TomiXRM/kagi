static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i10-s7",
    states: &["default", "focus-name"],
    size: (1100., 820.),
    render,
}];

fn main() {
    gallery_host::run(&PAGES);
}

fn render(cx: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    div()
        .size_full()
        .relative()
        .child(themes_page())
        .child(
            div()
                .absolute()
                .inset_0()
                .bg(Hsla::from(rgb(0x000000)).opacity(0.48)),
        )
        .child(theme_editor_sheet(cx.state == "focus-name"))
        .into_any_element()
}

        .justify_center()

        .child(
            div()
                .id("theme-editor-sheet")
                .w(pt(980.))
                .max_h_full()
                .flex()
                .flex_col()
                .rounded(pt(18.))
                .bg(rgb(ink::window()))
                .border_1()
                .border_color(rgb(ink::hairline()))
                .shadow_lg()
                .overflow_hidden()
                .child(sheet_header(name_has_focus))
                .child(
                    div().px(pt(20.)).py(pt(18.)).child(section(
                        "Colors",
                        None,
                        Some(
                            div()
                                .h(pt(metrics::CONTROL_HEIGHT))
                                .flex()
                                .items_center()
                                .gap(pt(8.))
                                .child("Preview")
                                .child(switch(
                                    "theme-preview",
                                    "Preview",
                                    true,
                                    Forced::Rest,
                                    |_, _, _| {},
                                ))
                                .into_any_element(),
                        ),
                        div()
                            .flex()
                            .flex_col()
                            .gap(pt(12.))
                            .child(group([
                                palette_group("Terminal", &palette.terminal, Forced::Rest),
                                palette_group("ANSI Standard", &palette.standard, Forced::Rest),
                                palette_group("ANSI Bright", &palette.bright, Forced::Rest),

fn palette_group(title: &'static str, swatches: &[Swatch], forced: Forced) -> AnyElement {
    div()
        .w_full()
        .px(pt(metrics::GROUP_PADDING_X))
        .py(pt(10.))
        .flex()
        .flex_col()
        .gap(pt(8.))
        .child(div().font_weight(FontWeight::SEMIBOLD).child(title))
        .child(
            div()
                .w_full()
                .flex()
                .items_start()
                .gap(pt(8.))
                .children(swatches.iter().map(|swatch| color_swatch(*swatch, forced))),
        )
        .into_any_element()
}

fn color_swatch(swatch: Swatch, forced: Forced) -> AnyElement {
    let hex = hex(swatch.color);
    ring(
        div()
            .id(format!("swatch-{}", swatch.id))
            .flex_1()
            .min_w_0()
            .flex()
            .flex_col()
            .gap(pt(4.))
            .child(
                div()
                    .text_size(pt(metrics::SECONDARY))
                    .text_color(ink::secondary())
                    .truncate()
                    .child(swatch.label),
            )
            .child(
                div()
                    .w_full()
                    .h(pt(metrics::CONTROL_HEIGHT))
                    .rounded(pt(metrics::CONTROL_RADIUS))
                    .bg(rgb(rgb_value(swatch.color)))
                    .border_1()
                    .border_color(rgb(ink::hairline())),
            )
            .child(
                div()
                    .w_full()
                    .text_size(pt(metrics::SECONDARY))
                    .text_color(ink::secondary())
                    .truncate()
                    .child(hex),
            ),
        forced == Forced::Focus,
        metrics::CONTROL_RADIUS,
    )
    .into_any_element()
}

fn theme_name_field(value: &'static str, forced: Forced) -> AnyElement {
    ring(
        div()
            .id("theme-name")
            .h(pt(30.))
            .px(pt(9.))
            .flex()
            .items_center()
            .rounded(pt(metrics::CONTROL_RADIUS))
            .bg(rgb(ink::control()))
            .border_1()
            .border_color(rgb(ink::hairline()))
            .when(forced == Forced::Hover, |el| {
                el.bg(rgb(ink::control_hover()))
            })
            .when(forced == Forced::Pressed, |el| {
                el.bg(rgb(ink::control_pressed()))
            })
            .when(forced == Forced::Disabled, |el| {
                el.opacity(ink::disabled_opacity())
            })
            .child(value),
        forced == Forced::Focus,
        metrics::CONTROL_RADIUS,
    )
    .into_any_element()
}

fn wide_action_button(id: &'static str, label: &'static str, forced: Forced) -> AnyElement {
    ring(
        div()
            .id(id)
            .w_full()
            .h(pt(metrics::CONTROL_HEIGHT))
            .flex()
            .items_center()
            .justify_center()
            .rounded(pt(metrics::CONTROL_RADIUS))
            .bg(rgb(ink::control()))
            .when(forced == Forced::Hover, |el| {
                el.bg(rgb(ink::control_hover()))
            })
            .when(forced == Forced::Pressed, |el| {
                el.bg(rgb(ink::control_pressed()))
            })
            .when(forced == Forced::Disabled, |el| {
                el.opacity(ink::disabled_opacity())
            })
            .child(label),
        forced == Forced::Focus,
        metrics::CONTROL_RADIUS,
    )
    .into_any_element()
}

fn primary_button(id: &'static str, label: &'static str, forced: Forced) -> AnyElement {
    let active = theme();
    ring(
        div()
            .id(id)
            .h(pt(metrics::CONTROL_HEIGHT))
            .px(pt(12.))
            .flex()
            .items_center()
            .rounded(pt(metrics::CONTROL_HEIGHT / 2.))
            .bg(rgb(active.ui_accent()))
            .text_color(Hsla::from(rgb(active.on_accent())))
            .when(forced == Forced::Disabled, |el| {
                el.opacity(ink::disabled_opacity())
            })
            .child(label),
        forced == Forced::Focus,
        metrics::CONTROL_HEIGHT / 2.,
    )
    .into_any_element()
}

fn rgb_value((red, green, blue): (u8, u8, u8)) -> u32 {
    ((red as u32) << 16) | ((green as u32) << 8) | blue as u32
}

fn hex(color: (u8, u8, u8)) -> String {
    format!("#{:02X}{:02X}{:02X}", color.0, color.1, color.2)
}
