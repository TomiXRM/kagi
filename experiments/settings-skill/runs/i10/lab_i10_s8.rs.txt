static PAGES: [GalleryPage; 1] = [GalleryPage {
    name: "lab-i10-s8",
    states: &[
        "default",
        "pane-appearance",
        "pane-graph",
        "pane-editor",
        "pane-smart-commit",
        "pane-terminal",
        "pane-network-sync",

    size: (1000., 860.),

}];

fn main() {
    gallery_host::run(&PAGES);
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Pane {
    Appearance,
    Graph,
    Editor,
    SmartCommit,
    Terminal,
    NetworkSync,
    General,
    Analyze,
}

impl Pane {
    const ALL: [Self; 8] = [
        Self::Appearance,
        Self::Graph,
        Self::Editor,
        Self::SmartCommit,
        Self::Terminal,
        Self::NetworkSync,
        Self::General,
        Self::Analyze,
    ];

    fn id(self) -> &'static str {
        match self {
            Self::Appearance => "appearance",
            Self::Graph => "graph",
            Self::Editor => "editor",
            Self::SmartCommit => "smart-commit",
            Self::Terminal => "terminal",
            Self::NetworkSync => "network-sync",
            Self::General => "general",
            Self::Analyze => "analyze",
        }
    }

    fn title(self) -> &'static str {
        match self {
            Self::Appearance => Msg::SettingsAppearance.t(),
            Self::Graph => Msg::SettingsGraph.t(),
            Self::Editor => Msg::ToolbarEditor.t(),
            Self::SmartCommit => Msg::SettingsSmartCommit.t(),
            Self::Terminal => Msg::ToolbarTerminal.t(),
            Self::NetworkSync => "Network & Sync",
            Self::General => Msg::SettingsGeneral.t(),
            Self::Analyze => Msg::SettingsAnalyze.t(),
        }
    }

    fn symbol(self) -> (IconName, Hue) {
        match self {
            Self::Appearance => (IconName::Palette, Hue::Indigo),
            Self::Graph => (IconName::GalleryVerticalEnd, Hue::Orange),
            Self::Editor => (IconName::File, Hue::Teal),
            Self::SmartCommit => (IconName::Bot, Hue::Purple),
            Self::Terminal => (IconName::SquareTerminal, Hue::Gray),
            Self::NetworkSync => (IconName::Network, Hue::Blue),
            Self::General => (IconName::Settings2, Hue::Gray),
            Self::Analyze => (IconName::ChartPie, Hue::Teal),
        }
    }
}

fn render(context: &GalleryContext, _: &mut Window, _: &mut App) -> AnyElement {
    let selected = Pane::ALL
        .into_iter()
        .find(|pane| context.state.strip_prefix("pane-") == Some(pane.id()))
        .unwrap_or(Pane::Appearance);
    let mut items = vec![search_field("search", "Search Settings").into_any_element()];
    for pane in Pane::ALL {
        let (icon, hue) = pane.symbol();
        items.push(
            sidebar_item(
                pane.id(),
                icon_tile(icon, Tint::Hue(hue), metrics::TILE),
                pane.title(),
                None,
                selected == pane,
                Forced::Rest,
            )
            .into_any_element(),
        );
    }
    let sections = match selected {
        Pane::Appearance => appearance(),
        Pane::Graph => graph(),
        Pane::Editor => editor(),
        Pane::SmartCommit => smart_commit(),
        Pane::Terminal => terminal(),
        Pane::NetworkSync => network(),
        Pane::General => general(context.lang),
        Pane::Analyze => analyze(),
    };
    settings_window(
        sidebar(items),
        detail_page("pane-scroll", selected.title(), sections),
    )
    .into_any_element()
}

fn toggle(id: &'static str, title: Msg, subtitle: Msg, checked: bool) -> AnyElement {
    Row::new(title.t())
        .subtitle(subtitle.t())
        .trailing(switch(id, title.t(), checked, Forced::Rest, |_, _, _| {}))
        .into_any_element()
}

/// Kit gap: multiline text field, shown as a gallery fixture (no editing).
/// Unlike a trailing control this is a full-width editor, not a row accessory.
/// There are no positioned selection/caret indicators in its rest state.
fn multiline_value(value: &str, forced: Forced) -> gpui::Stateful<Div> {
    let fill = match forced {
        Forced::Hover => ink::control_hover(),
        Forced::Pressed => ink::control_pressed(),
        _ => ink::control(),
    };
    ring(
        div()
            .id("ignore-patterns")
            .w_full()
            .min_w_0()
            .min_h(pt(150.))
            .p(pt(10.))
            .flex()
            .flex_col()
            .gap(pt(5.))
            .bg(rgb(fill))
            .text_color(ink::text())
            .text_size(pt(metrics::BODY))
            .aria_label(Msg::A11yAnalyzeIgnorePatterns.t())
            .children(value.lines().map(|line| {
                div()
                    .child(SharedString::from(line.to_owned()))
                    .into_any_element()
            })),
        forced == Forced::Focus,
        metrics::CONTROL_RADIUS,
    )
}
